# MyASR Implementation Plan

> **For Claude:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an offline Russian-language speech-recognition Windows desktop app (WinForms, .NET 10) on top of GigaAM v3 RNN-T ONNX models, with push-to-talk capture, VU meter, global hotkey, and auto-download of model files.

**Architecture:** Layered modules behind interfaces — `IMicCapture`, `IFeatureExtractor`, `IRnnTDecoder`, `IModelProvider`, `IHotkeyManager`, `IAsrService` — composed via a hand-rolled `Microsoft.Extensions.DependencyInjection` container in `Program.cs`. Heavy ONNX sessions are created once and reused; recognition runs on the thread pool, UI updates marshal back through `Form.BeginInvoke`. Bottom-up build order so each layer is unit-tested before the next is added.

**Tech Stack:** .NET 10 WinForms, C# 13, ONNX Runtime (CPU), NAudio.Wasapi, Microsoft.Extensions.DependencyInjection / Logging, TUnit + TUnit.Assertions for tests, Python (one-shot tooling) for golden-reference generation.

**Specification:** [`docs/superpowers/specs/2026-05-09-asr-gigaam-design.md`](../specs/2026-05-09-asr-gigaam-design.md)

**Conventions:**
- One source file per type. Files under ~150 lines whenever possible.
- File-scoped namespaces, `nullable enable`, primary constructors, collection expressions.
- Conventional Commits (`feat:`, `test:`, `chore:`, `docs:`, `refactor:`, `fix:`).
- Every task ends with a single commit. `git status` must be clean before moving on.
- TDD: red → green → commit. Refactor only when tests are green.
- Reference skills as needed:

**Resolved open items from the spec:**

```csharp
using System.IO;

namespace MyASR.Asr;

public sealed class Vocabulary
{
    private const string BlankToken = "<blk>";
    private const string SpaceMarker = "▁";

    private readonly string[] _tokens;

    private Vocabulary(string[] tokens)
    {
        _tokens = tokens;
        BlankIndex = Array.IndexOf(tokens, BlankToken);
        if (BlankIndex < 0)
            throw new InvalidDataException($"Vocabulary must contain '{BlankToken}'.");
    }

    public int Count => _tokens.Length;
    public int BlankIndex { get; }
    public string this[int index] => _tokens[index];

    public static Vocabulary Load(string path)
    {
        var lines = File.ReadAllLines(path);
        var tokens = new string[lines.Length];
        for (var i = 0; i < lines.Length; i++)
        {
            // Format: "<token> <index>" — index is implicit by line order.
            var space = lines[i].LastIndexOf(' ');
            tokens[i] = space < 0 ? lines[i] : lines[i][..space];
        }
        return new Vocabulary(tokens);
    }

    public string Decode(IReadOnlyList<int> ids)
    {
        var sb = new System.Text.StringBuilder(ids.Count);
        foreach (var id in ids)
        {
            if (id == BlankIndex) continue;
            var t = _tokens[id];
            sb.Append(t == SpaceMarker ? " " : t);
        }
        return sb.ToString();
    }
}
```

```python
"""
Generate golden reference artefacts for MyASR.Tests:
- reference_filterbank_64x161.json
- reference.wav (16kHz mono PCM)
- reference_mel.bin  (T x 64 float32, row-major)
- reference.txt      (expected transcription)

Run once: python tools/generate_reference.py
Requires: numpy, soundfile, gigaam (pip install gigaam).
"""
import json
import math
import os
import struct
import sys

import numpy as np

OUT_DIR = os.path.join(os.path.dirname(__file__), "..", "MyASR.Tests", "Assets")
os.makedirs(OUT_DIR, exist_ok=True)

SR = 16000
N_FFT = 320
HOP = 160
N_MELS = 64

def hz_to_mel(f):
    return 2595.0 * math.log10(1.0 + f / 700.0)

def mel_to_hz(m):
    return 700.0 * (10 ** (m / 2595.0) - 1.0)

def htk_filterbank(sr=SR, n_fft=N_FFT, n_mels=N_MELS):
    n_freqs = n_fft // 2 + 1
    f_low_mel = hz_to_mel(0.0)
    f_high_mel = hz_to_mel(sr / 2)
    mel_pts = np.linspace(f_low_mel, f_high_mel, n_mels + 2)
    hz_pts = np.array([mel_to_hz(m) for m in mel_pts])
    bin_pts = np.floor((n_fft + 1) * hz_pts / sr).astype(int)

    fb = np.zeros((n_mels, n_freqs), dtype=np.float64)
    for m in range(1, n_mels + 1):
        f_m_minus = bin_pts[m - 1]
        f_m       = bin_pts[m]
        f_m_plus  = bin_pts[m + 1]
        for k in range(f_m_minus, f_m):
            if f_m == f_m_minus: continue
            fb[m - 1, k] = (k - f_m_minus) / (f_m - f_m_minus)
        for k in range(f_m, f_m_plus):
            if f_m_plus == f_m: continue
            fb[m - 1, k] = (f_m_plus - k) / (f_m_plus - f_m)
    return fb

def write_filterbank_json(fb):
    data = {"shape": list(fb.shape), "data": fb.flatten().tolist()}
    with open(os.path.join(OUT_DIR, "reference_filterbank_64x161.json"), "w", encoding="utf-8") as f:
        json.dump(data, f)

def write_reference_audio_and_mel():
    """Synthesizes a deterministic chirp + recognised text using gigaam."""
    try:
        import gigaam
    except ImportError:
        print("gigaam not installed — skipping reference.wav/mel/txt; run pip install gigaam first.")
        return

    import soundfile as sf
    # 3-sec chirp 200 Hz → 4000 Hz at 16 kHz, plus 1 sec silence (so encoder sees real signal).
    t = np.linspace(0, 3, 3 * SR, endpoint=False)
    chirp = 0.3 * np.sin(2 * np.pi * (200 + (4000 - 200) * t / 3) * t)
    audio = np.concatenate([chirp, np.zeros(SR, dtype=np.float32)]).astype(np.float32)
    wav_path = os.path.join(OUT_DIR, "reference.wav")
    sf.write(wav_path, audio, SR, subtype="PCM_16")

    # Compute mel through gigaam for byte-exact comparison.
    model = gigaam.load_model("v3_rnnt")
    fe = model.preprocessor
    import torch
    feats = fe(torch.from_numpy(audio).unsqueeze(0)).squeeze(0).numpy()  # [n_mels, T]
    feats_t_mels = feats.T.astype(np.float32, copy=False)                # [T, n_mels]

    mel_path = os.path.join(OUT_DIR, "reference_mel.bin")
    with open(mel_path, "wb") as f:
        f.write(struct.pack("<II", feats_t_mels.shape[0], feats_t_mels.shape[1]))
        f.write(feats_t_mels.tobytes())

    text = model.recognize(wav_path).strip()
    with open(os.path.join(OUT_DIR, "reference.txt"), "w", encoding="utf-8") as f:
        f.write(text)

if __name__ == "__main__":
    fb = htk_filterbank()
    write_filterbank_json(fb)
    write_reference_audio_and_mel()
    print("done")
```

```csharp
namespace MyASR.Features;

public static class MelFilterBank
{
    public static double[,] BuildHtk(int sampleRate, int nFft, int nMels)
    {
        var nFreqs = nFft / 2 + 1;
        var fLowMel  = HzToMel(0.0);
        var fHighMel = HzToMel(sampleRate / 2.0);

        var melPts = new double[nMels + 2];
        for (var i = 0; i < melPts.Length; i++)
            melPts[i] = fLowMel + (fHighMel - fLowMel) * i / (melPts.Length - 1);

        var binPts = new int[melPts.Length];
        for (var i = 0; i < melPts.Length; i++)
        {
            var hz = MelToHz(melPts[i]);
            binPts[i] = (int)Math.Floor((nFft + 1) * hz / sampleRate);
        }

        var fb = new double[nMels, nFreqs];
        for (var m = 1; m <= nMels; m++)
        {
            int left = binPts[m - 1], center = binPts[m], right = binPts[m + 1];
            for (var k = left; k < center; k++)
                if (center != left)
                    fb[m - 1, k] = (k - left) / (double)(center - left);
            for (var k = center; k < right; k++)
                if (right != center)
                    fb[m - 1, k] = (right - k) / (double)(right - center);
        }
        return fb;
    }

    private static double HzToMel(double f) => 2595.0 * Math.Log10(1.0 + f / 700.0);
    private static double MelToHz(double m) => 700.0 * (Math.Pow(10.0, m / 2595.0) - 1.0);
}
```

`MyASR/Features/PowerSpectrum.cs`:

```csharp
using NAudio.Dsp;

namespace MyASR.Features;

internal static class PowerSpectrum
{
    /// <summary>
    /// Computes |FFT(x ∘ window)|² for a single frame and writes <c>nFft/2 + 1</c> bins to <paramref name="output"/>.
    /// Length of <paramref name="frame"/> must equal <paramref name="nFft"/>.
    /// </summary>
    public static void Compute(ReadOnlySpan<float> frame, ReadOnlySpan<float> window, int nFft, Span<float> output)
    {
        if (frame.Length != nFft) throw new ArgumentException("frame length != nFft", nameof(frame));
        if (window.Length != nFft) throw new ArgumentException("window length != nFft", nameof(window));
        if (output.Length != nFft / 2 + 1) throw new ArgumentException("output length != nFft/2+1", nameof(output));

        var m = (int)Math.Log2(nFft);
        if (1 << m != nFft) throw new ArgumentException("nFft must be power of 2 for NAudio FFT", nameof(nFft));

        Span<Complex> buf = stackalloc Complex[nFft];
        for (var i = 0; i < nFft; i++)
        {
            buf[i].X = frame[i] * window[i];
            buf[i].Y = 0f;
        }
        FastFourierTransform.FFT(true, m, buf);
        for (var k = 0; k <= nFft / 2; k++)
            output[k] = buf[k].X * buf[k].X + buf[k].Y * buf[k].Y;
    }

    public static float[] HanningWindow(int n)
    {
        var w = new float[n];
        for (var i = 0; i < n; i++)
            w[i] = (float)(0.5 - 0.5 * Math.Cos(2 * Math.PI * i / (n - 1)));
        return w;
    }
}
```

> **Caveat:** `n_fft = 320` is **not** a power of 2, but NAudio's FFT requires it. Two options:
> 1. Zero-pad to 512 — changes spectrum, breaks byte-exact match with gigaam.
> 2. Use a different FFT implementation that supports arbitrary length.
>
> We pick option 2 in Task 3.2 and switch `PowerSpectrum` over to `MathNet.Numerics.IntegralTransforms.Fourier`. For now the test uses sin and silence which would still pass a 512 zero-padded FFT, so we keep NAudio temporarily — but the byte-exact test in Task 3.2 will fail and that's the trigger to swap implementations.

```csharp
namespace MyASR.Features;

public sealed class MelSpectrogramExtractor : IFeatureExtractor
{
    private const int SampleRate = 16000;
    private const int NFft       = 320;
    private const int HopLength  = 160;
    private const int NMels      = 64;
    private const float LogOffset = 1e-9f;

    private readonly float[] _window;
    private readonly double[,] _filterBank;

    public MelSpectrogramExtractor()
    {
        _window     = PowerSpectrum.HanningWindow(NFft);
        _filterBank = MelFilterBank.BuildHtk(SampleRate, NFft, NMels);
    }

    public float[,] Extract(ReadOnlySpan<float> pcm)
    {
        if (pcm.Length < NFft) return new float[0, NMels];
        var t = (pcm.Length - NFft) / HopLength + 1;
        var feats = new float[t, NMels];

        Span<float> spec = stackalloc float[NFft / 2 + 1];

        for (var i = 0; i < t; i++)
        {
            var offset = i * HopLength;
            PowerSpectrum.Compute(pcm.Slice(offset, NFft), _window, NFft, spec);
            for (var m = 0; m < NMels; m++)
            {
                double sum = 0;
                for (var k = 0; k < spec.Length; k++)
                    sum += _filterBank[m, k] * spec[k];
                feats[i, m] = (float)Math.Log(sum + LogOffset);
            }
        }
        return feats;
    }
}
```

```csharp
namespace MyASR.Audio;

/// <summary>
/// Single-producer, single-consumer ring buffer of <see cref="float"/> samples.
/// Snapshot returns the last <c>≤Capacity</c> samples in FIFO order.
/// Older samples are discarded silently when the writer overruns.
/// </summary>
public sealed class AudioRingBuffer(int capacitySamples)
{
    private readonly float[] _buffer = new float[capacitySamples];
    private long _writePos;            // monotonic, writer-only
    private readonly Lock _gate = new();

    public int Capacity => _buffer.Length;

    public void Write(ReadOnlySpan<float> data)
    {
        lock (_gate)
        {
            foreach (var s in data)
            {
                _buffer[(int)(_writePos % _buffer.Length)] = s;
                _writePos++;
            }
        }
    }

    public float[] Snapshot()
    {
        lock (_gate)
        {
            var n = (int)Math.Min(_writePos, _buffer.Length);
            if (n == 0) return [];
            var output = new float[n];
            var startWrap = (int)((_writePos - n) % _buffer.Length);
            var first = Math.Min(n, _buffer.Length - startWrap);
            Array.Copy(_buffer, startWrap, output, 0, first);
            if (first < n)
                Array.Copy(_buffer, 0, output, first, n - first);
            return output;
        }
    }

    public void Reset()
    {
        lock (_gate) { _writePos = 0; Array.Clear(_buffer); }
    }
}
```

```csharp
using NAudio.Wave;
using NAudio.Wave.SampleProviders;

namespace MyASR.Audio;

/// <summary>
/// Converts arbitrary <see cref="WaveFormat"/> float PCM into 16 kHz mono float.
/// Used between WASAPI capture and the feature extractor.
/// </summary>
public static class Resampler
{
    public static float[] ToMono16k(WaveFormat src, ReadOnlySpan<float> samples)
    {
        if (src.SampleRate == 16000 && src.Channels == 1)
            return samples.ToArray();

        // Build an in-memory ISampleProvider stream from the raw float buffer.
        var ms = new MemoryStream();
        var bw = new BinaryWriter(ms);
        foreach (var s in samples) bw.Write(s);
        bw.Flush();
        ms.Position = 0;

        var floatFormat = WaveFormat.CreateIeeeFloatWaveFormat(src.SampleRate, src.Channels);
        var raw = new RawSourceWaveStream(ms, floatFormat).ToSampleProvider();

        ISampleProvider mono = src.Channels == 1
            ? raw
            : new StereoToMonoSampleProvider(raw) { LeftVolume = 0.5f, RightVolume = 0.5f };

        var resampled = src.SampleRate == 16000
            ? mono
            : new WdlResamplingSampleProvider(mono, 16000);

        var output = new List<float>(samples.Length / 3);
        var buffer = new float[8192];
        int n;
        while ((n = resampled.Read(buffer, 0, buffer.Length)) > 0)
            output.AddRange(buffer.AsSpan(0, n).ToArray());
        return [.. output];
    }
}
```

```csharp
using System.Diagnostics;
using System.IO;
using System.Net.Http;
using System.Net.Http.Headers;

namespace MyASR.Models;

public sealed class HuggingFaceModelProvider(HttpClient http, IReadOnlyList<ModelFileEntry> manifest)
    : IModelProvider
{
    public bool AreModelsAvailable(ModelPaths paths)
    {
        foreach (var f in manifest)
            if (!File.Exists(Path.Combine(paths.Root, f.FileName))) return false;
        return true;
    }

    public async Task EnsureDownloadedAsync(ModelPaths paths, IProgress<DownloadProgress>? progress, CancellationToken ct)
    {
        Directory.CreateDirectory(paths.Root);
        foreach (var f in manifest)
        {
            var dst = Path.Combine(paths.Root, f.FileName);
            if (File.Exists(dst)) continue;
            await DownloadOneAsync(f, dst, progress, ct);
        }
    }

    private async Task DownloadOneAsync(ModelFileEntry f, string dst, IProgress<DownloadProgress>? progress, CancellationToken ct)
    {
        var partial = dst + ".partial";
        long resumeFrom = File.Exists(partial) ? new FileInfo(partial).Length : 0;

        using var req = new HttpRequestMessage(HttpMethod.Get, f.Url);
        if (resumeFrom > 0)
            req.Headers.Range = new RangeHeaderValue(resumeFrom, null);

        using var resp = await http.SendAsync(req, HttpCompletionOption.ResponseHeadersRead, ct);
        resp.EnsureSuccessStatusCode();

        var total = (resp.Content.Headers.ContentLength ?? f.Size) + resumeFrom;
        await using var input  = await resp.Content.ReadAsStreamAsync(ct);
        await using var output = new FileStream(partial, FileMode.OpenOrCreate, FileAccess.Write, FileShare.None);
        output.Position = resumeFrom;

        var sw = Stopwatch.StartNew();
        var buffer = new byte[81920];
        long downloaded = resumeFrom;
        long lastReportedAt = downloaded;
        var lastReportTime = sw.Elapsed;

        int n;
        while ((n = await input.ReadAsync(buffer, ct)) > 0)
        {
            await output.WriteAsync(buffer.AsMemory(0, n), ct);
            downloaded += n;

            var now = sw.Elapsed;
            if (progress != null && (now - lastReportTime).TotalMilliseconds >= 250)
            {
                var bps = (downloaded - lastReportedAt) / (now - lastReportTime).TotalSeconds;
                progress.Report(new DownloadProgress(f.FileName, downloaded, total, bps));
                lastReportedAt = downloaded;
                lastReportTime = now;
            }
        }
        await output.FlushAsync(ct);
        output.Close();

        if (f.Sha256 is not null && !VerifySha(partial, f.Sha256))
        {
            File.Delete(partial);
            throw new InvalidDataException($"SHA-256 mismatch for {f.FileName}");
        }
        File.Move(partial, dst, overwrite: true);
        progress?.Report(new DownloadProgress(f.FileName, total, total, 0));
    }

    private static bool VerifySha(string path, string expectedHex)
    {
        using var sha = System.Security.Cryptography.SHA256.Create();
        using var fs  = File.OpenRead(path);
        var actual = Convert.ToHexString(sha.ComputeHash(fs)).ToLowerInvariant();
        return actual.Equals(expectedHex.ToLowerInvariant(), StringComparison.Ordinal);
    }
}
```

```csharp
using Microsoft.ML.OnnxRuntime;
using Microsoft.ML.OnnxRuntime.Tensors;
using MyASR.Models;
using System.Diagnostics;

namespace MyASR.Asr;

/// <summary>
/// Greedy RNN-T decoder. Loads three ONNX sessions and runs the standard
/// encoder → (decoder + joint) loop emitting tokens whenever joint &gt; blank.
/// </summary>
public sealed class RnnTGreedyDecoder : IRnnTDecoder
{
    private const int MaxSymbolsPerStep = 5;

    private readonly InferenceSession _encoder;
    private readonly InferenceSession _decoder;
    private readonly InferenceSession _joint;
    private readonly Vocabulary _vocab;

    public RnnTGreedyDecoder(ModelPaths paths, Vocabulary vocab)
    {
        var opts = new SessionOptions
        {
            ExecutionMode = ExecutionMode.ORT_PARALLEL,
            GraphOptimizationLevel = GraphOptimizationLevel.ORT_ENABLE_ALL,
            InterOpNumThreads = 1,
            IntraOpNumThreads = Math.Max(1, Environment.ProcessorCount / 2)
        };
        _encoder = new InferenceSession(paths.Encoder, opts);
        _decoder = new InferenceSession(paths.Decoder, opts);
        _joint   = new InferenceSession(paths.Joint,   opts);
        _vocab   = vocab;
    }

    public AsrResult Decode(float[,] mel, TimeSpan audioDuration)
    {
        var sw = Stopwatch.StartNew();
        var tokens = new List<int>(64);

        // 1) Encoder: input [1, n_mels=64, T] → [1, d_model=768, T/4]
        var encOut = RunEncoder(mel);

        // 2) Greedy loop over encoder timesteps.
        var (decState1, decState2) = InitDecoderState();
        var lastToken = _vocab.BlankIndex;
        var tFrames = encOut.GetLength(2);

        for (var t = 0; t < tFrames; t++)
        {
            var encT = SliceTime(encOut, t);
            var emittedAtT = 0;
            while (emittedAtT < MaxSymbolsPerStep)
            {
                var (predOut, ns1, ns2) = RunDecoder(lastToken, decState1, decState2);
                var logits = RunJoint(encT, predOut);
                var token = ArgMax(logits);
                if (token == _vocab.BlankIndex) break;
                tokens.Add(token);
                lastToken = token;
                decState1 = ns1; decState2 = ns2;
                emittedAtT++;
            }
        }

        var text = _vocab.Decode(tokens);
        return new AsrResult(text, audioDuration, sw.Elapsed);
    }

    // --- ONNX plumbing -----------------------------------------------------

    private float[,,] RunEncoder(float[,] mel)
    {
        var T = mel.GetLength(0);
        var M = mel.GetLength(1);
        var inputData = new float[1 * M * T];
        // [1, M, T] in row-major: outer=batch=1, mid=mels, inner=time.
        for (var t = 0; t < T; t++)
            for (var m = 0; m < M; m++)
                inputData[m * T + t] = mel[t, m];

        var input = new DenseTensor<float>(inputData, [1, M, T]);
        var lengths = new DenseTensor<long>([T], [1]);

        var inputs = new[]
        {
            NamedOnnxValue.CreateFromTensor(_encoder.InputMetadata.Keys.First(), input),
            NamedOnnxValue.CreateFromTensor(_encoder.InputMetadata.Keys.Skip(1).First(), lengths)
        };
        using var results = _encoder.Run(inputs);
        var firstOut = results.First().AsTensor<float>(); // [1, 768, T/4]
        var dims = firstOut.Dimensions.ToArray();
        var arr = firstOut.ToArray();
        var outT = dims[2];
        var outD = dims[1];
        var output = new float[1, outD, outT];
        for (var d = 0; d < outD; d++)
            for (var t = 0; t < outT; t++)
                output[0, d, t] = arr[d * outT + t];
        return output;
    }

    private float[] SliceTime(float[,,] enc, int t)
    {
        var d = enc.GetLength(1);
        var slice = new float[d];
        for (var i = 0; i < d; i++) slice[i] = enc[0, i, t];
        return slice;
    }

    private (DenseTensor<float>, DenseTensor<float>) InitDecoderState()
    {
        // 1 LSTM layer, hidden=320, batch=1.
        var z = new float[1 * 1 * 320];
        return (new DenseTensor<float>(z, [1, 1, 320]),
                new DenseTensor<float>(z.ToArray(), [1, 1, 320]));
    }

    private (float[] predHidden, DenseTensor<float>, DenseTensor<float>) RunDecoder(
        int lastToken, DenseTensor<float> h, DenseTensor<float> c)
    {
        var input = new DenseTensor<long>([lastToken], [1, 1]);
        var inLen = new DenseTensor<long>([1L], [1]);
        var inputs = new[]
        {
            NamedOnnxValue.CreateFromTensor(_decoder.InputMetadata.Keys.ElementAt(0), input),
            NamedOnnxValue.CreateFromTensor(_decoder.InputMetadata.Keys.ElementAt(1), inLen),
            NamedOnnxValue.CreateFromTensor(_decoder.InputMetadata.Keys.ElementAt(2), h),
            NamedOnnxValue.CreateFromTensor(_decoder.InputMetadata.Keys.ElementAt(3), c)
        };
        using var results = _decoder.Run(inputs).ToArray();
        var pred = results[0].AsTensor<float>().ToArray(); // [1, 320, 1]
        var newH = new DenseTensor<float>(results[1].AsTensor<float>().ToArray(), [1, 1, 320]);
        var newC = new DenseTensor<float>(results[2].AsTensor<float>().ToArray(), [1, 1, 320]);
        return (pred, newH, newC);
    }

    private float[] RunJoint(float[] encStep, float[] predHidden)
    {
        var enc = new DenseTensor<float>(encStep, [1, encStep.Length, 1]);
        var pred = new DenseTensor<float>(predHidden, [1, predHidden.Length, 1]);
        var inputs = new[]
        {
            NamedOnnxValue.CreateFromTensor(_joint.InputMetadata.Keys.ElementAt(0), enc),
            NamedOnnxValue.CreateFromTensor(_joint.InputMetadata.Keys.ElementAt(1), pred)
        };
        using var results = _joint.Run(inputs);
        return results.First().AsTensor<float>().ToArray();
    }

    private static int ArgMax(ReadOnlySpan<float> v)
    {
        var idx = 0; var best = v[0];
        for (var i = 1; i < v.Length; i++) if (v[i] > best) { best = v[i]; idx = i; }
        return idx;
    }

    public void Dispose()
    {
        _encoder.Dispose();
        _decoder.Dispose();
        _joint.Dispose();
    }
}
```

> Tensor names are accessed via `InputMetadata.Keys.ElementAt(...)` so we don't hard-code names that may differ between gigaam exports. Once the model is downloaded, run `python tools/inspect_model.py` (one-liner with `onnx.load`) to confirm the order: encoder = `[features, lengths]`; decoder = `[targets, lengths, h_state, c_state]`; joint = `[enc, pred]`. If any model uses different ordering, adjust the indexes here.

```csharp
using NAudio.CoreAudioApi;
using NAudio.Wave;

namespace MyASR.Audio;

public sealed class WasapiMicCapture : IMicCapture
{
    private const int RingCapacity = 16000 * 31; // 31 sec headroom

    private readonly AudioRingBuffer _ring = new(RingCapacity);
    private WasapiCapture? _capture;
    private double _rmsDb = -120.0;

    public IReadOnlyList<DeviceInfo> EnumerateDevices()
    {
        using var en = new MMDeviceEnumerator();
        return [.. en.EnumerateAudioEndPoints(DataFlow.Capture, DeviceState.Active)
                     .Select(d => new DeviceInfo(d.ID, d.FriendlyName))];
    }

    public double CurrentRmsDb => _rmsDb;

    public void Start(string deviceId)
    {
        Stop();
        using var en = new MMDeviceEnumerator();
        var device = en.GetDevice(deviceId);
        _capture = new WasapiCapture(device);
        _capture.DataAvailable += OnData;
        _ring.Reset();
        _capture.StartRecording();
    }

    public float[] StopAndDrain()
    {
        Stop();
        return _ring.Snapshot();
    }

    private void OnData(object? sender, WaveInEventArgs e)
    {
        if (_capture is null) return;
        var fmt = _capture.WaveFormat;
        var samples = BytesToFloat(fmt, e.Buffer.AsSpan(0, e.BytesRecorded));
        var mono16 = Resampler.ToMono16k(fmt, samples);
        _ring.Write(mono16);
        _rmsDb = ComputeRmsDb(mono16);
    }

    private static float[] BytesToFloat(WaveFormat fmt, ReadOnlySpan<byte> bytes)
    {
        // WASAPI shared mode delivers IEEE float by default; if 16-bit PCM, convert.
        if (fmt.Encoding == WaveFormatEncoding.IeeeFloat)
        {
            var floats = new float[bytes.Length / 4];
            System.Buffers.Binary.BinaryPrimitives.GetType(); // ensure System.Buffers referenced
            for (var i = 0; i < floats.Length; i++)
                floats[i] = BitConverter.ToSingle(bytes.Slice(i * 4, 4));
            return floats;
        }
        if (fmt.Encoding == WaveFormatEncoding.Pcm && fmt.BitsPerSample == 16)
        {
            var floats = new float[bytes.Length / 2];
            for (var i = 0; i < floats.Length; i++)
            {
                var sample = (short)(bytes[i * 2] | (bytes[i * 2 + 1] << 8));
                floats[i] = sample / 32768f;
            }
            return floats;
        }
        throw new NotSupportedException($"Unsupported capture format: {fmt}");
    }

    private static double ComputeRmsDb(ReadOnlySpan<float> samples)
    {
        if (samples.IsEmpty) return -120.0;
        double sum = 0;
        foreach (var s in samples) sum += s * s;
        var rms = Math.Sqrt(sum / samples.Length);
        return 20.0 * Math.Log10(Math.Max(rms, 1e-10));
    }

    private void Stop()
    {
        if (_capture is null) return;
        _capture.DataAvailable -= OnData;
        _capture.StopRecording();
        _capture.Dispose();
        _capture = null;
    }

    public void Dispose() => Stop();
}
```

```csharp
    : IAsrService, IDisposable

    private const int MinSamples = 16000 * 3 / 10; // 0.3 sec @ 16 kHz

    public AsrState State { get; private set; } = AsrState.Idle;

    public event Action<AsrStateChanged>? StateChanged;
    public event Action<AsrResult>? ResultReady;
    public event Action<Exception>? Failed;

    public void StartCapture(string deviceId)
    {
        if (State != AsrState.Idle) return;
        mic.Start(deviceId);
        SetState(AsrState.Recording);
    }

    public async Task StopAndRecognizeAsync()
    {
        if (State != AsrState.Recording) return;
        var pcm = mic.StopAndDrain();
        if (pcm.Length < MinSamples) { SetState(AsrState.Idle); return; }

        SetState(AsrState.Decoding);
        try
        {
            var result = await Task.Run(() =>
            {
                var feats = features.Extract(pcm);
                return decoder.Decode(feats, TimeSpan.FromSeconds(pcm.Length / 16000.0));
            });
            ResultReady?.Invoke(result);
        }
        catch (Exception ex)
        {
            Failed?.Invoke(ex);
        }
        finally
        {
            SetState(AsrState.Idle);
        }
    }

    private void SetState(AsrState s)
    {
        if (State == s) return;
        State = s;
        StateChanged?.Invoke(new AsrStateChanged(s));
    }

    public void Dispose()
    {
        mic.Dispose();
        decoder.Dispose();
    }
}
```

```csharp
namespace MyASR.Hotkeys;

public sealed class WindowsHotkeyManager : IHotkeyManager
{
    private record Registered(IntPtr Hwnd, int Id);
    private readonly List<Registered> _registered = [];

    [LibraryImport("user32.dll", SetLastError = true)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static partial bool RegisterHotKey(IntPtr hWnd, int id, uint fsModifiers, uint vk);

    [LibraryImport("user32.dll", SetLastError = true)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static partial bool UnregisterHotKey(IntPtr hWnd, int id);

    [LibraryImport("user32.dll")]
    private static partial short GetAsyncKeyState(int vKey);

    public bool Register(IntPtr hWnd, int id, uint modifiers, uint vk)
    {
        if (RegisterHotKey(hWnd, id, modifiers, vk))
        {
            _registered.Add(new Registered(hWnd, id));
            return true;
        }
        return false;
    }

    public bool IsKeyDown(uint vk) => (GetAsyncKeyState((int)vk) & 0x8000) != 0;

    public void Dispose()
    {
        foreach (var r in _registered) UnregisterHotKey(r.Hwnd, r.Id);
        _registered.Clear();
    }
}
```

```csharp
namespace MyASR;

public partial class Form1 : Form
{

    private const int WM_HOTKEY = 0x0312;

    private readonly IServiceProvider _sp;
    private readonly IAsrService _asr;
    private readonly IMicCapture _mic;
    private readonly IModelProvider _modelProvider;
    private readonly ModelPaths _paths;
    private readonly IHotkeyManager _hotkeys;
    private readonly ILogger<Form1>? _logger;

    private readonly Stopwatch _recordingTimer = new();
    private readonly System.Windows.Forms.Timer _maxLengthTimer = new() { Interval = 30_000 };
    private bool _hotkeyDownPrev;

    public Form1(
        IServiceProvider sp,
        IAsrService asr,
        IMicCapture mic,
        IModelProvider modelProvider,
        ModelPaths paths,
        IHotkeyManager hotkeys,

        _micButton.MouseDown += (_, _) => StartRecording();
        _micButton.MouseUp   += async (_, _) => await StopRecording();
        _micButton.MouseLeave += async (_, _) => { if (_asr.State == AsrState.Recording) await StopRecording(); };

        _clearBtn.Click   += (_, _) => _resultBox.Clear();
        _copyBtn.Click    += (_, _) => CopyToClipboard();
        _saveTxtBtn.Click += (_, _) => SaveAsTxt();

        Load   += async (_, _) => await OnFormLoadAsync();
        FormClosed += (_, _) => Cleanup();

    private async Task OnFormLoadAsync()
    {
        PopulateDevices();

        if (!_modelProvider.AreModelsAvailable(_paths))
        {
            using var dlg = _sp.GetRequiredService<DownloadForm>();
            if (dlg.ShowDialog(this) != DialogResult.OK)
            {
                MessageBox.Show("Модели не загружены — приложение будет закрыто.", "MyASR", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                Close();
                return;
            }

        _hotkeys.Register(Handle, HotkeyId, 0, 0x78); // VK_F9
        _hotkeyPoll.Start();

        await Task.CompletedTask;

        base.WndProc(ref m);
```

```csharp
        Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
        "MyASR", "logs");
    Directory.CreateDirectory(logDir);
    var path = Path.Combine(logDir, $"app-{DateTime.Now:yyyyMMdd}.log");

private sealed class SimpleFileLoggerProvider(TextWriter w) : ILoggerProvider
{
    public ILogger CreateLogger(string categoryName) => new SimpleFileLogger(w, categoryName);
    public void Dispose() => w.Dispose();
}

private sealed class SimpleFileLogger(TextWriter w, string category) : ILogger
{
    public IDisposable BeginScope<TState>(TState state) where TState : notnull => NullScope.Instance;
    public bool IsEnabled(LogLevel logLevel) => logLevel >= LogLevel.Information;
    public void Log<TState>(LogLevel level, EventId id, TState state, Exception? ex, Func<TState, Exception?, string> formatter)
    {
        if (!IsEnabled(level)) return;
        lock (w) w.WriteLine($"{DateTime.Now:HH:mm:ss.fff} {level} {category} :: {formatter(state, ex)} {ex}");
    }
    private sealed class NullScope : IDisposable { public static readonly NullScope Instance = new(); public void Dispose() { } }
}
```

```csharp
using System.Text.Json;

namespace MyASR;

public sealed class UserSettings
{
    public string? MicDeviceId { get; set; }
    public string Hotkey { get; set; } = "F9";
    public bool GlobalHotkey { get; set; } = true;

    private static string Path =>
        System.IO.Path.Combine(
            Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
            "MyASR", "settings.json");

    public static UserSettings Load()
    {
        try
        {
            if (File.Exists(Path))
                return JsonSerializer.Deserialize<UserSettings>(File.ReadAllText(Path)) ?? new();
        }
        catch { /* swallow — corrupt file → defaults */ }
        return new UserSettings();
    }

    public void Save()
    {
        Directory.CreateDirectory(System.IO.Path.GetDirectoryName(Path)!);
        File.WriteAllText(Path, JsonSerializer.Serialize(this, new JsonSerializerOptions { WriteIndented = true }));
    }
}
```

## Risks & Mitigations

| Risk | Mitigation |
|---|---|
| ONNX tensor names change between gigaam exports | `RnnTGreedyDecoder` resolves names via `InputMetadata.Keys.ElementAt(...)`; document expected ordering. |
| `n_fft = 320` not power-of-two breaks NAudio FFT | Task 3.1 Step 7 swaps to `MathNet.Numerics`. |
| Mel parity below tolerance | Task 3.2 byte-exact test acts as a tripwire. |
| WASAPI device formats vary | `Resampler` handles arbitrary input via `WdlResamplingSampleProvider`. |
| Long downloads on slow connections | `HuggingFaceModelProvider` resumes with `Range:` header; UI offers retry. |
| Confluent threading bugs | All UI updates funnel through `BeginInvoke`; ringbuffer is `lock`-protected; AsrService FSM forbids re-entry. |
